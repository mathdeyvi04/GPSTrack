104 static void
105 degrees_to_NMEA(
106 double graus_decimais,
107 bool is_lat,
108 std::string& ddmm,
109 char& hemisf
110 ){
111 hemisf = (is_lat) ? (
112 ( graus_decimais >= 0 ) ? 'N' : 'S'
113 ) :
114 (
115 ( graus_decimais >= 0 ) ? 'E' : 'W'
116 );
117
118 double valor_abs = std::fabs(graus_decimais);
119 int graus = static_cast<int>(std::floor(valor_abs));
120 double min = (valor_abs - graus) * 60.0;
121
122 // Não utilizamos apenas a função de formatar_inteiro, pois
123 // utilizaremos o oss em seguida.
124 std::ostringstream oss;
125 oss << std::setfill('0')
126 << std::setw(is_lat ? 2 : 3)
127 << graus
128 << std::fixed
129 << std::setprecision(4)
130 << std::setw(7)
131 << std::right
132 << min;
133
134 ddmm = oss.str();
135 }

346 GPSSim(
347 double latitude_inicial_graus,
348 double longitude_inicial_graus,

356
357 // Cria o par de pseudo-terminais
358 if(
359 ::openpty( &fd_pai, &fd_filho, caminho_do_pseudo_terminal, nullptr, nullptr ) != 0
360 ){
361 throw std::runtime_error("Falha ao criar pseudo-terminal");
362 }
363
364 // Configura o terminal filho para simular o módulo real (9600 8N1)
365 termios config_com{}; // Cria a estrutura vazia
366 ::tcgetattr(fd_filho, &config_com); // Lê as configurações atuais e armazena na struct
367 ::cfsetispeed(&config_com, B9600); // Definimos velocidade de entrada e de saída
368 ::cfsetospeed(&config_com, B9600); // Essa constante está presente dentro do termios.h
369 // Diversas operações bits a bits
370 config_com.c_cflag = (config_com.c_cflag & ~CSIZE) | CS8;
371 config_com.c_cflag |= (CLOCAL | CREAD);
372 config_com.c_cflag &= ~(PARENB | CSTOPB);
373 config_com.c_iflag = IGNPAR;
374 config_com.c_oflag = 0;
375 config_com.c_lflag = 0;
376 tcsetattr(fd_filho, TCSANOW, &config_com); // Aplicamos as configurações
377
378 // Fecha o filho - será aberto pelo usuário no caminho correto
379 // Mantemos o Pai aberto para procedimentos posteriores
380 ::close(fd_filho);
381 }
